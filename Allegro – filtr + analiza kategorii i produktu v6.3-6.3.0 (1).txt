// ==UserScript==
// @name         Allegro – filtr + analiza kategorii i produktu v6.3
// @namespace    natan-allegro
// @version      6.3.0
// @description  Ręczny filtr ofert, analiza kategorii i analiza produktu bez ciągłych pętli.
// @match        https://allegro.pl/*
// @match        https://www.allegro.pl/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==

(() => {
    'use strict';

    const PREFIX = 'natan-allegro-v63-';

    const DEFAULTS = {
        activeTab: 'category',
        collapsed: false,

        categoryMinSales: 50,
        maxCategoryOffers: 10000,
        minStrongOffers: 3,

        filterMinSales: 0,
        hideSponsored: false,
        hideUnknown: false
    };

    const state = {
        activeTab: load(
            'activeTab',
            DEFAULTS.activeTab
        ),

        collapsed: loadBool(
            'collapsed',
            DEFAULTS.collapsed
        ),

        categoryMinSales: loadNum(
            'categoryMinSales',
            DEFAULTS.categoryMinSales,
            1,
            9999
        ),

        maxCategoryOffers: loadNum(
            'maxCategoryOffers',
            DEFAULTS.maxCategoryOffers,
            100,
            999999
        ),

        minStrongOffers: loadNum(
            'minStrongOffers',
            DEFAULTS.minStrongOffers,
            1,
            100
        ),

        filterMinSales: loadNum(
            'filterMinSales',
            DEFAULTS.filterMinSales,
            0,
            999
        ),

        hideSponsored: loadBool(
            'hideSponsored',
            DEFAULTS.hideSponsored
        ),

        hideUnknown: loadBool(
            'hideUnknown',
            DEFAULTS.hideUnknown
        )
    };

    const CLS = {
        hiddenSales:
            'natan-hidden-sales-v63',

        hiddenSponsored:
            'natan-hidden-sponsored-v63',

        hiddenUnknown:
            'natan-hidden-unknown-v63'
    };

    let lastCategory = null;
    let lastProduct = null;
    let lastFilter = null;

    function key(name) {
        return PREFIX + name;
    }

    function clamp(
        value,
        minimum,
        maximum
    ) {
        return Math.min(
            maximum,
            Math.max(
                minimum,
                value
            )
        );
    }

    function load(
        name,
        fallback
    ) {
        return (
            localStorage.getItem(
                key(name)
            ) ??
            fallback
        );
    }

    function loadBool(
        name,
        fallback
    ) {
        const value =
            localStorage.getItem(
                key(name)
            );

        return value === null
            ? fallback
            : value === 'true';
    }

    function loadNum(
        name,
        fallback,
        minimum,
        maximum
    ) {
        const value = Number(
            localStorage.getItem(
                key(name)
            )
        );

        return Number.isFinite(value)
            ? clamp(
                Math.round(value),
                minimum,
                maximum
            )
            : fallback;
    }

    function save() {
        Object.entries(state)
            .forEach(
                ([name, value]) => {
                    localStorage.setItem(
                        key(name),
                        String(value)
                    );
                }
            );
    }

    function reset() {
        Object.keys(localStorage)
            .filter(
                item =>
                    item.startsWith(
                        PREFIX
                    )
            )
            .forEach(
                item =>
                    localStorage.removeItem(
                        item
                    )
            );

        clearFilter();
        location.reload();
    }

    function norm(value) {
        return String(
            value ?? ''
        )
            .replace(
                /\u00a0/g,
                ' '
            )
            .replace(
                /\s+/g,
                ' '
            )
            .trim()
            .toLowerCase();
    }

    function esc(value) {
        return String(
            value ?? ''
        )
            .replaceAll(
                '&',
                '&amp;'
            )
            .replaceAll(
                '<',
                '&lt;'
            )
            .replaceAll(
                '>',
                '&gt;'
            )
            .replaceAll(
                '"',
                '&quot;'
            )
            .replaceAll(
                "'",
                '&#039;'
            );
    }

    function fmt(
        value,
        digits = 0
    ) {
        return Number.isFinite(value)
            ? value.toLocaleString(
                'pl-PL',
                {
                    minimumFractionDigits:
                        digits,

                    maximumFractionDigits:
                        digits
                }
            )
            : '—';
    }

    function avg(values) {
        if (!values.length) {
            return 0;
        }

        return Math.round(
            values.reduce(
                (
                    sum,
                    value
                ) =>
                    sum + value,
                0
            ) /
            values.length
        );
    }

    function parseCompact(raw) {
        const text = String(
            raw ?? ''
        )
            .replace(
                /\u00a0/g,
                ''
            )
            .replace(
                /\s+/g,
                ''
            )
            .replace(
                ',',
                '.'
            )
            .toLowerCase();

        let match = text.match(
            /^(\d+(?:\.\d+)?)tys\.?\+?$/
        );

        if (match) {
            return Math.round(
                Number(
                    match[1]
                ) * 1000
            );
        }

        match = text.match(
            /^(\d+(?:\.\d+)?)mln\.?\+?$/
        );

        if (match) {
            return Math.round(
                Number(
                    match[1]
                ) * 1000000
            );
        }

        const digits =
            text.match(/\d+/g);

        return digits
            ? Number(
                digits.join('')
            )
            : null;
    }

    function parseSalesText(text) {
    const normalized = norm(text);

    /*
     * Obsługiwane:
     * 1,2 tys. osób kupiło
     * 1,2 tys. osoby kupiły
     * 1,2 tys. osoba kupiła
     */
    let match = normalized.match(
        /(\d+(?:[.,]\d+)?)\s*tys\.?\s*(?:osób|osoby|osoba)\s+kupi\w*/
    );

    if (match) {
        const value = Number(
            match[1].replace(',', '.')
        );

        return Number.isFinite(value)
            ? Math.round(value * 1000)
            : null;
    }

    /*
     * Obsługiwane:
     * 340 osób kupiło
     * 192 osoby kupiły
     * 1 osoba kupiła
     */
    match = normalized.match(
        /(\d[\d\s]*)\s*(?:osób|osoby|osoba)\s+kupi\w*/
    );

    if (!match) {
        return null;
    }

    const value = Number(
        match[1].replace(/\s/g, '')
    );

    return Number.isFinite(value)
        ? value
        : null;
}

    function getOfferArticles() {
        return [
            ...document
                .querySelectorAll(
                    'article'
                )
        ].filter(
            article => {
                const hasTitle =
                    Boolean(
                        article.querySelector(
                            'h2 a'
                        )
                    );

                const hasOffer =
                    Boolean(
                        article.querySelector(
                            [
                                'a[href*="/oferta/"]',
                                'a[href*="/events/clicks"]',
                                '[data-role-type="add-to-cart-button"]'
                            ].join(',')
                        )
                    );

                return (
                    hasTitle &&
                    hasOffer
                );
            }
        );
    }

    function visualContainer(
        article
    ) {
        const listItem =
            article.closest('li');

        if (
            listItem &&
            listItem
                .querySelectorAll(
                    ':scope article'
                )
                .length === 1
        ) {
            return listItem;
        }

        return article;
    }

    function offerSales(article) {
        const ariaElements =
            article.querySelectorAll(
                '[aria-label]'
            );

        for (
            const element
            of ariaElements
        ) {
            const label =
                element.getAttribute(
                    'aria-label'
                ) || '';

            if (
                norm(label)
                    .includes('kupi')
            ) {
                const sales =
                    parseSalesText(
                        label
                    );

                if (sales !== null) {
                    return sales;
                }
            }
        }

        return parseSalesText(
            article.textContent || ''
        );
    }

    function offerSponsored(article) {
        const labels =
            article.querySelectorAll(
                'p, span, div'
            );

        for (
            const element
            of labels
        ) {
            const text =
                norm(
                    element.textContent
                );

            if (
                text ===
                    'sponsorowane' ||
                text ===
                    'promowane' ||
                text ===
                    'oferta sponsorowana'
            ) {
                return true;
            }
        }

        return Boolean(
            article.querySelector(
                [
                    'a[href*="/events/clicks"][href*="bi_s%3Dads"]',
                    'a[href*="/events/clicks"][href*="bi_s=ads"]'
                ].join(',')
            )
        );
    }

    function offerRecords() {
        return getOfferArticles()
            .map(
                article => {
                    const sales =
                        offerSales(
                            article
                        );

                    const sponsored =
                        offerSponsored(
                            article
                        );

                    article.dataset
                        .natanSales =
                        sales === null
                            ? 'unknown'
                            : String(sales);

                    article.dataset
                        .natanSponsored =
                        String(
                            sponsored
                        );

                    return {
                        article,

                        container:
                            visualContainer(
                                article
                            ),

                        sales,
                        sponsored
                    };
                }
            );
    }

    function clearFilter() {
        document
            .querySelectorAll(
                [
                    `.${CLS.hiddenSales}`,
                    `.${CLS.hiddenSponsored}`,
                    `.${CLS.hiddenUnknown}`
                ].join(',')
            )
            .forEach(
                element => {
                    element.classList
                        .remove(
                            CLS.hiddenSales,
                            CLS.hiddenSponsored,
                            CLS.hiddenUnknown
                        );
                }
            );
    }

    function applyFilter() {
        const startedAt =
            performance.now();

        clearFilter();

        const records =
            offerRecords();

        let visible = 0;
        let low = 0;
        let sponsored = 0;
        let unknown = 0;

        for (
            const record
            of records
        ) {
            if (
                record.sponsored &&
                state.hideSponsored
            ) {
                record.container
                    .classList.add(
                        CLS.hiddenSponsored
                    );

                sponsored++;
                continue;
            }

            if (
                record.sales === null
            ) {
                if (
                    state.hideUnknown
                ) {
                    record.container
                        .classList.add(
                            CLS.hiddenUnknown
                        );

                    unknown++;
                } else {
                    visible++;
                }

                continue;
            }

            if (
                record.sales <
                state.filterMinSales
            ) {
                record.container
                    .classList.add(
                        CLS.hiddenSales
                    );

                low++;
                continue;
            }

            visible++;
        }

        lastFilter = {
            total:
                records.length,

            visible,
            low,
            sponsored,
            unknown,

            ms:
                Math.round(
                    performance.now() -
                    startedAt
                )
        };

        renderFilter();
    }

    function removeFilter() {
        clearFilter();

        const amount =
            getOfferArticles()
                .length;

        lastFilter = {
            total:
                amount,

            visible:
                amount,

            low:
                0,

            sponsored:
                0,

            unknown:
                0,

            ms:
                0
        };

        renderFilter();
    }

    function categoryName() {
        return (
            document
                .querySelector('h1')
                ?.textContent || ''
        ).trim() ||
        'Nie wykryto';
    }

    function categoryOfferCount() {
        const heading =
            document.querySelector(
                'h1'
            );

        const roots =
            heading?.parentElement
                ? [
                    heading.parentElement,
                    document.body
                ]
                : [
                    document.body
                ];

        for (
            const root
            of roots
        ) {
            const candidates =
                root.querySelectorAll(
                    'div, span, p'
                );

            for (
                const element
                of candidates
            ) {
                const match =
                    norm(
                        element.textContent
                    ).match(
                        /^([\d\s.,]+(?:tys\.?|mln\.?)?\+?)\s*ofert(?:a|y)?$/
                    );

                if (match) {
                    const amount =
                        parseCompact(
                            match[1]
                        );

                    if (
                        amount !== null
                    ) {
                        return amount;
                    }
                }
            }
        }

        return null;
    }

    function analyzeCategory() {
        const startedAt =
            performance.now();

        const records =
            offerRecords();

        const allSales = [];
        const organicSales = [];
        const sponsoredSales = [];

        let unknown = 0;
        let sponsoredCount = 0;

        for (
            const record
            of records
        ) {
            if (
                record.sponsored
            ) {
                sponsoredCount++;
            }

            if (
                record.sales === null
            ) {
                unknown++;
                continue;
            }

            allSales.push(
                record.sales
            );

            if (
                record.sponsored
            ) {
                sponsoredSales.push(
                    record.sales
                );
            } else {
                organicSales.push(
                    record.sales
                );
            }
        }

        allSales.sort(
            (
                first,
                second
            ) =>
                second - first
        );

        organicSales.sort(
            (
                first,
                second
            ) =>
                second - first
        );

        sponsoredSales.sort(
            (
                first,
                second
            ) =>
                second - first
        );

        const strong =
            allSales.filter(
                value =>
                    value >=
                    state
                        .categoryMinSales
            );

        const result = {
            name:
                categoryName(),

            count:
                categoryOfferCount(),

            detected:
                records.length,

            withSales:
                allSales.length,

            organic:
                organicSales.length,

            promoted:
                sponsoredSales.length,

            sponsoredCount,

            unknown,

            strong:
                strong.length,

            strongOrganic:
                organicSales.filter(
                    value =>
                        value >=
                        state
                            .categoryMinSales
                ).length,

            strongPromoted:
                sponsoredSales.filter(
                    value =>
                        value >=
                        state
                            .categoryMinSales
                ).length,

            max:
                allSales[0] || 0,

            top5:
                avg(
                    allSales.slice(
                        0,
                        5
                    )
                ),

            ms:
                Math.round(
                    performance.now() -
                    startedAt
                )
        };

        result.score =
            categoryScore(
                result
            );

        result.verdict =
            categoryVerdict(
                result.score
            );

        return result;
    }

    function categoryScore(
        analysis
    ) {
        let score = 0;

        if (
            analysis.count !== null
        ) {
            if (
                analysis.count <=
                3000
            ) {
                score += 30;
            } else if (
                analysis.count <=
                6000
            ) {
                score += 26;
            } else if (
                analysis.count <=
                state
                    .maxCategoryOffers
            ) {
                score += 20;
            } else if (
                analysis.count <=
                state
                    .maxCategoryOffers *
                    1.5
            ) {
                score += 10;
            }
        }

        if (
            analysis.strong >= 10
        ) {
            score += 40;
        } else if (
            analysis.strong >= 7
        ) {
            score += 37;
        } else if (
            analysis.strong >= 5
        ) {
            score += 34;
        } else if (
            analysis.strong >= 3
        ) {
            score += 28;
        } else if (
            analysis.strong === 2
        ) {
            score += 18;
        } else if (
            analysis.strong === 1
        ) {
            score += 9;
        }

        if (
            analysis.strong >=
            state.minStrongOffers
        ) {
            score += 15;
        }

        if (
            analysis.top5 >=
            state.categoryMinSales *
            3
        ) {
            score += 10;
        } else if (
            analysis.top5 >=
            state.categoryMinSales *
            2
        ) {
            score += 8;
        } else if (
            analysis.top5 >=
            state.categoryMinSales
        ) {
            score += 4;
        }

        if (
            analysis.max >=
            state.categoryMinSales *
            5
        ) {
            score += 5;
        } else if (
            analysis.max >=
            state.categoryMinSales *
            3
        ) {
            score += 4;
        } else if (
            analysis.max >=
            state.categoryMinSales *
            2
        ) {
            score += 2;
        }

        return clamp(
            score,
            0,
            100
        );
    }

    function categoryVerdict(
        score
    ) {
        if (
            score >= 85
        ) {
            return {
                label:
                    'BANGER',

                cls:
                    'banger',

                desc:
                    'Bardzo mocna kategoria.'
            };
        }

        if (
            score >= 65
        ) {
            return {
                label:
                    'DOBRA',

                cls:
                    'good',

                desc:
                    'Kategoria wygląda dobrze.'
            };
        }

        if (
            score >= 45
        ) {
            return {
                label:
                    'DO SPRAWDZENIA',

                cls:
                    'medium',

                desc:
                    'Kategoria wymaga dalszej weryfikacji.'
            };
        }

        return {
            label:
                'SŁABA',

            cls:
                'bad',

            desc:
                'Za mało popytu albo za duża konkurencja.'
        };
    }

    function runCategory() {
        const button =
            document.getElementById(
                'natan-category-run'
            );

        button.disabled = true;
        button.textContent =
            'Analizowanie...';

        requestAnimationFrame(
            () => {
                lastCategory =
                    analyzeCategory();

                renderCategory();
                renderAdvanced();

                button.disabled =
                    false;

                button.textContent =
                    'Analizuj kategorię';
            }
        );
    }

    function productBrand() {
        const element =
            document.querySelector(
                [
                    '[data-analytics-view-label="brandLink"]',
                    '[data-analytics-click-label="parameterLink"][data-analytics-click-value="Marka"]'
                ].join(',')
            );

        if (
            element?.dataset
                .analyticsViewValue
        ) {
            return element
                .dataset
                .analyticsViewValue
                .trim();
        }

        const candidate =
            [
                ...document
                    .querySelectorAll(
                        'div, p, span, td'
                    )
            ].find(
                item =>
                    /^marka:\s*/i.test(
                        (
                            item.textContent ||
                            ''
                        ).trim()
                    )
            );

        const text =
            candidate
                ?.textContent ||
            '';

        return text
            .replace(
                /^marka:\s*/i,
                ''
            )
            .trim() ||
            'Nie wykryto';
    }

    function productTitle() {
        return (
            document
                .querySelector('h1')
                ?.textContent || ''
        ).trim() ||
        'Nie wykryto';
    }

    function productRating() {
        const element =
            document.querySelector(
                '[data-analytics-view-label="productRating"]'
            );

        const value = Number(
            element?.getAttribute(
                'data-analytics-view-custom-rating-value'
            )
        );

        if (
            Number.isFinite(value)
        ) {
            return value;
        }

        const aria =
            document
                .querySelector(
                    '[aria-label^="ocena:"]'
                )
                ?.getAttribute(
                    'aria-label'
                ) || '';

        const match =
            aria.match(
                /ocena:\s*([\d,.]+)/i
            );

        return match
            ? Number(
                match[1]
                    .replace(
                        ',',
                        '.'
                    )
            )
            : null;
    }

    function productReviewCounts() {
        const element =
            document.querySelector(
                '[data-analytics-view-label="productRating"]'
            );

        const ratingCount =
            Number(
                element?.getAttribute(
                    'data-analytics-view-custom-rating-count'
                )
            );

        const text =
            element?.textContent ||
            document.body.innerText;

        const ratings =
            Number.isFinite(
                ratingCount
            )
                ? ratingCount
                : Number(
                    (
                        text.match(
                            /(\d[\d\s]*)\s+ocen/i
                        )?.[1] ||
                        ''
                    ).replace(
                        /\s/g,
                        ''
                    )
                ) || 0;

        const reviews =
            Number(
                (
                    text.match(
                        /(\d[\d\s]*)\s+recenzj/i
                    )?.[1] ||
                    ''
                ).replace(
                    /\s/g,
                    ''
                )
            ) || 0;

        return {
            ratings,
            reviews
        };
    }

function productSales() {
    const ratingLink = document.querySelector(
        '[data-analytics-view-label="productRating"]'
    );

    if (ratingLink) {
        let current = ratingLink.parentElement;

        for (let level = 0; level < 4 && current; level++) {
            const text = current.textContent || '';

            if (
                /(?:osób|osoby|osoba)\s+kupi\w*\s+ostatnio/i.test(text)
            ) {
                const sales = parseSalesText(text);

                if (sales !== null) {
                    return sales;
                }
            }

            current = current.parentElement;
        }
    }

    const candidates = [
        ...document.querySelectorAll('div, span, p')
    ];

    for (const element of candidates) {
        const text = norm(
            element.textContent
        );

        if (
            !text.includes('ostatnio') ||
            !/(?:osób|osoby|osoba)\s+kupi\w*/.test(text)
        ) {
            continue;
        }

        const sales = parseSalesText(text);

        if (sales !== null) {
            return sales;
        }
    }

    return null;
}

    function currentOfferSales() {
        const text =
            document.body
                .innerText || '';

        const match =
            text.match(
                /(\d[\d\s]*)\s+(?:osób|osoby|osoba)\s+kupiło\s+tę\s+ofertę/i
            );

        return match
            ? Number(
                match[1]
                    .replace(
                        /\s/g,
                        ''
                    )
            )
            : null;
    }

    function catalogueOffers() {
    const link = document.querySelector(
        'a.other-offers-link-all[href="#inne-oferty-produktu"]'
    );

    if (!link) {
        return null;
    }

    const text = link.textContent || '';

    const match = text.match(
        /wszystkie\s+oferty\s*\((\d[\d\s]*)\)/i
    );

    if (!match) {
        return null;
    }

    const value = Number(
        match[1].replace(/\s/g, '')
    );

    return Number.isFinite(value)
        ? value
        : null;
}

    function officialShopInfo() {
        let label = '';

        const elements =
            document.querySelectorAll(
                'div, p, span'
            );

        for (
            const element
            of elements
        ) {
            const text =
                (
                    element
                        .textContent ||
                    ''
                ).trim();

            if (
                /^od\s+oficjalny sklep/i
                    .test(text)
            ) {
                label = text;
                break;
            }
        }

        const logo =
            document.querySelector(
                'img[alt*="logo oficjalnego sklepu" i]'
            );

        if (
            !label &&
            logo
        ) {
            label =
                logo.getAttribute(
                    'alt'
                ) || '';
        }

        const official =
            Boolean(
                label ||
                logo
            );

        const shop =
            label
                .replace(
                    /^od\s+oficjalny sklep\s*/i,
                    ''
                )
                .replace(
                    /^logo oficjalnego sklepu\s*/i,
                    ''
                )
                .trim();

        return {
            official,

            shop:
                shop ||
                'Nie wykryto'
        };
    }

    function sameBrandShop(
        brand,
        shop
    ) {
        const first =
            norm(brand)
                .replace(
                    /[^a-z0-9ąćęłńóśźż]/g,
                    ''
                );

        const second =
            norm(shop)
                .replace(
                    /[^a-z0-9ąćęłńóśźż]/g,
                    ''
                );

        return Boolean(
            first &&
            second &&
            (
                first === second ||
                first.includes(
                    second
                ) ||
                second.includes(
                    first
                )
            )
        );
    }

    function productPrice() {
        const candidates = [
            document
                .querySelector(
                    '[data-analytics-interaction-custom-price]'
                )
                ?.getAttribute(
                    'data-analytics-interaction-custom-price'
                ),

            document
                .querySelector(
                    '[data-analytics-view-label="price"]'
                )
                ?.textContent
        ];

        for (
            const raw
            of candidates
        ) {
            if (!raw) {
                continue;
            }

            const direct =
                Number(
                    String(raw)
                        .replace(
                            ',',
                            '.'
                        )
                );

            if (
                Number.isFinite(
                    direct
                ) &&
                direct > 0
            ) {
                return direct;
            }

            const match =
                String(raw)
                    .match(
                        /(\d[\d\s]*)[,.]([0-9]{2})\s*zł/i
                    );

            if (match) {
                return Number(
                    match[1]
                        .replace(
                            /\s/g,
                            ''
                        ) +
                    '.' +
                    match[2]
                );
            }
        }

        const body =
            document.body
                .innerText || '';

        const matches =
            [
                ...body.matchAll(
                    /(\d[\d\s]*)[,.]([0-9]{2})\s*zł/g
                )
            ]
                .map(
                    match =>
                        Number(
                            match[1]
                                .replace(
                                    /\s/g,
                                    ''
                                ) +
                            '.' +
                            match[2]
                        )
                )
                .filter(
                    value =>
                        value > 1
                );

        return matches.length
            ? matches[0]
            : null;
    }

    function roundUp89(price) {
        let candidate =
            Math.floor(
                price / 10
            ) *
            10 +
            9.89;

        if (
            candidate < price
        ) {
            candidate += 10;
        }

        return Math.round(
            candidate * 100
        ) / 100;
    }

    function pricing(
        purchase
    ) {
        if (
            !Number.isFinite(
                purchase
            )
        ) {
            return null;
        }

        const commission =
            0.12;

        const tax =
            0.19;

        const targetNet =
            100;

        const profitBeforeTax =
            targetNet /
            (
                1 - tax
            );

        const minimum =
            (
                purchase +
                profitBeforeTax
            ) /
            (
                1 - commission
            );

        const suggested =
            roundUp89(
                minimum
            );

        const commissionValue =
            suggested *
            commission;

        const grossProfitBeforeTax =
            suggested -
            commissionValue -
            purchase;

        const incomeTax =
            Math.max(
                0,
                grossProfitBeforeTax *
                tax
            );

        const netProfit =
            grossProfitBeforeTax -
            incomeTax;

        return {
            purchase,
            commission,
            tax,
            targetNet,
            minimum,
            suggested,
            commissionValue,
            profitBeforeTax:
                grossProfitBeforeTax,
            incomeTax,
            netProfit
        };
    }

    function productScore(
        analysis
    ) {
        if (
            analysis.sales === null ||
            analysis.sales < 50
        ) {
            return 0;
        }

        if (
            analysis.official &&
            analysis.brandMatchesShop &&
            (
                analysis.offers ?? 0
            ) < 8
        ) {
            return 0;
        }

        let score = 0;

        if (
            analysis.sales >= 300
        ) {
            score += 35;
        } else if (
            analysis.sales >= 200
        ) {
            score += 31;
        } else if (
            analysis.sales >= 100
        ) {
            score += 27;
        } else {
            score += 22;
        }

        if (
            analysis.rating !== null
        ) {
            if (
                analysis.rating >= 4.95
            ) {
                score += 15;
            } else if (
                analysis.rating >= 4.9
            ) {
                score += 13;
            } else if (
                analysis.rating >= 4.8
            ) {
                score += 10;
            } else if (
                analysis.rating >= 4.7
            ) {
                score += 6;
            }
        }

        if (
            analysis.ratings >= 500
        ) {
            score += 15;
        } else if (
            analysis.ratings >= 250
        ) {
            score += 13;
        } else if (
            analysis.ratings >= 100
        ) {
            score += 10;
        } else if (
            analysis.ratings >= 50
        ) {
            score += 7;
        } else if (
            analysis.ratings >= 20
        ) {
            score += 4;
        }

        if (
            analysis.reviews >= 100
        ) {
            score += 10;
        } else if (
            analysis.reviews >= 50
        ) {
            score += 8;
        } else if (
            analysis.reviews >= 20
        ) {
            score += 6;
        } else if (
            analysis.reviews >= 10
        ) {
            score += 4;
        }

        if (
            analysis.offers !== null
        ) {
            if (
                analysis.offers >= 20 &&
                analysis.offers <= 60
            ) {
                score += 15;
            } else if (
                analysis.offers >= 9 &&
                analysis.offers < 20
            ) {
                score += 12;
            } else if (
                analysis.offers > 60 &&
                analysis.offers <= 120
            ) {
                score += 10;
            } else if (
                analysis.offers >= 8
            ) {
                score += 7;
            }
        }


        return clamp(
            score,
            0,
            100
        );
    }

    function productVerdict(
        analysis
    ) {
        if (
            analysis.sales === null
        ) {
            return {
                label:
                    'BRAK DANYCH',

                cls:
                    'medium',

                desc:
                    'Nie wykryto sprzedaży produktu.'
            };
        }

        if (
            analysis.sales < 50
        ) {
            return {
                label:
                    'ODRZUĆ',

                cls:
                    'bad',

                desc:
                    `Sprzedaż produktu wynosi ${analysis.sales}; wymagane minimum to 50.`
            };
        }

        if (
            analysis.official &&
            analysis.brandMatchesShop &&
            (
                analysis.offers ?? 0
            ) < 8
        ) {
            return {
                label:
                    'ODRZUĆ',

                cls:
                    'bad',

                desc:
                    'Oficjalny sklep tej samej marki i mniej niż 8 ofert w katalogu.'
            };
        }

        if (
            analysis.score >= 85
        ) {
            return {
                label:
                    'BANGER',

                cls:
                    'banger',

                desc:
                    'Bardzo mocny produkt.'
            };
        }

        if (
            analysis.score >= 65
        ) {
            return {
                label:
                    'DOBRY',

                cls:
                    'good',

                desc:
                    'Produkt wygląda dobrze.'
            };
        }

        if (
            analysis.score >= 45
        ) {
            return {
                label:
                    'Średni',

                cls:
                    'medium',

                desc:
                    'Prawdopodbnie mały popyt'
            };
        }

        return {
            label:
                'ODRZUĆ',

            cls:
                'bad',

            desc:
                'Produkt nie uzyskał wystarczającej oceny.'
        };
    }

    function analyzeProduct() {
        const startedAt =
            performance.now();

        const brand =
            productBrand();

        const officialInfo =
            officialShopInfo();

        const counts =
            productReviewCounts();

        const result = {
            title:
                productTitle(),

            brand,

            official:
                officialInfo.official,

            shop:
                officialInfo.shop,

            brandMatchesShop:
                sameBrandShop(
                    brand,
                    officialInfo.shop
                ),

            offers:
                catalogueOffers(),

            sales:
                productSales(),

            rating:
                productRating(),

            ratings:
                counts.ratings,

            reviews:
                counts.reviews,

            price:
                productPrice(),

            ms:
                0
        };

        result.pricing =
            pricing(
                result.price
            );

        result.score =
            productScore(
                result
            );

        result.verdict =
            productVerdict(
                result
            );

        result.ms =
            Math.round(
                performance.now() -
                startedAt
            );

        return result;
    }
